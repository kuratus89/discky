cmake_minimum_required(VERSION 3.16)

project(discky
    VERSION 1.0.0
    LANGUAGES C
)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

add_library(discky STATIC
    src/discky.c
    src/window.c
    src/memory.c
    src/render.c
    src/screen.c
)

set_target_properties(discky PROPERTIES
    ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib
)

target_include_directories(discky
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

add_executable(hello example/first.c)
target_link_libraries(hello PRIVATE discky)
