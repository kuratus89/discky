cmake_minimum_required(VERSION 3.16)

project(discky
    VERSION 1.0.0
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)


set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

add_library(discky STATIC
    src/discky.cpp
    src/window.cpp
    # src/memory.cpp
    src/render.cpp
    src/screen.cpp
)

set_target_properties(discky PROPERTIES
    ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib
)

target_include_directories(discky
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

add_executable(rectangle example/rectangle.cpp)
add_executable(animate_rectangle example/animate_rectangle.cpp)
add_executable(animate_circle example/animate_circle.cpp)
add_executable(all_shape example/all_shapes.cpp)
add_executable(diamond example/diamond.cpp)
add_executable(pacman example/pacman.cpp)
add_executable(text example/text.cpp)
target_link_libraries(rectangle PRIVATE discky)
target_link_libraries(animate_rectangle PRIVATE discky)
target_link_libraries(animate_circle PRIVATE discky)
target_link_libraries(all_shape PRIVATE discky)
target_link_libraries(diamond PRIVATE discky)
target_link_libraries(text PRIVATE discky)
target_link_libraries(pacman PRIVATE discky)
